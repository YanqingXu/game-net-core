cmake_minimum_required(VERSION 3.20)
project(GameNetTimerAttribution LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Requires Linux Release")
endif()
find_package(GameNetCore 0.3.1 EXACT CONFIG REQUIRED COMPONENTS experimental_io_uring)
foreach(target IN ITEMS idle timer_idle_trace timer_network_trace)
    if(target STREQUAL "timer_network_trace")
        add_executable(${target} ../installed_acceptance/acceptance.cpp observer.cpp)
    elseif(target STREQUAL "timer_idle_trace")
        add_executable(${target} idle.cpp observer.cpp)
    else()
        add_executable(${target} idle.cpp)
    endif()
    target_compile_features(${target} PRIVATE cxx_std_23)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    target_link_libraries(${target} PRIVATE GameNet::core GameNet::experimental_io_uring)
    if(NOT target STREQUAL "idle")
        target_compile_definitions(${target} PRIVATE GAMENET_TIMER_DIAGNOSTICS=1)
        target_link_options(${target} PRIVATE -Wl,--wrap=epoll_wait -Wl,--wrap=epoll_pwait2)
    endif()
endforeach()
