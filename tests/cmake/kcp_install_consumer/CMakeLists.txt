cmake_minimum_required(VERSION 3.20)
project(GameNetKcpConsumer LANGUAGES CXX)
set(MODE "kcp" CACHE STRING "kcp, udp or core")
if(MODE STREQUAL "kcp")
    find_package(GameNetCore CONFIG REQUIRED COMPONENTS kcp)
elseif(MODE STREQUAL "udp")
    find_package(GameNetCore CONFIG REQUIRED COMPONENTS udp)
else()
    find_package(GameNetCore CONFIG REQUIRED)
endif()
foreach(forbidden OpenSSL::SSL GameNet::tls GameNet::rpc_tls GameNet::websocket GameNet::websocket_tls GameNet::udp_transport)
    if(TARGET ${forbidden})
        message(FATAL_ERROR "Unexpected optional dependency: ${forbidden}")
    endif()
endforeach()
if(NOT MODE STREQUAL "kcp" AND TARGET GameNet::kcp)
    message(FATAL_ERROR "KCP imported by Core/UDP-only discovery")
endif()
if(MODE STREQUAL "core" AND TARGET GameNet::udp)
    message(FATAL_ERROR "UDP imported by Core-only discovery")
endif()
add_executable(kcp_consumer main.cpp)
target_link_libraries(kcp_consumer PRIVATE GameNet::core)
if(MODE STREQUAL "kcp")
    target_link_libraries(kcp_consumer PRIVATE GameNet::kcp)
    target_sources(kcp_consumer PRIVATE ../../integration/kcp/reference_source.cc)
    target_compile_definitions(kcp_consumer PRIVATE CONSUMER_KCP=1)
elseif(MODE STREQUAL "udp")
    target_link_libraries(kcp_consumer PRIVATE GameNet::udp)
endif()
enable_testing()
add_test(NAME kcp_installed_consumer COMMAND kcp_consumer)
