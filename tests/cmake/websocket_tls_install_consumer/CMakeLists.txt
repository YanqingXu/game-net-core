cmake_minimum_required(VERSION 3.20)
project(WssConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_EXTENSIONS OFF)
if(REQUEST_COMPONENT STREQUAL "core")
    find_package(GameNetCore REQUIRED)
else()
    find_package(GameNetCore REQUIRED COMPONENTS ${REQUEST_COMPONENT})
endif()
if(REQUEST_COMPONENT STREQUAL "core" OR REQUEST_COMPONENT STREQUAL "websocket")
    if(TARGET GameNet::tls OR TARGET GameNet::websocket_tls OR OpenSSL_FOUND)
        message(FATAL_ERROR "Core/WS-only discovery leaked TLS")
    endif()
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE GameNet::${REQUEST_COMPONENT})
if(REQUEST_COMPONENT STREQUAL "websocket_tls")
    target_compile_definitions(consumer PRIVATE CONSUME_WSS)
elseif(REQUEST_COMPONENT STREQUAL "websocket")
    target_compile_definitions(consumer PRIVATE CONSUME_WS)
endif()
enable_testing()
add_test(NAME installed.consumer COMMAND consumer)
