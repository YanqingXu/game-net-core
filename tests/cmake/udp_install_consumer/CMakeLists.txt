cmake_minimum_required(VERSION 3.20)
project(GameNetUdpConsumer LANGUAGES CXX)
option(REQUEST_UDP "Request optional UDP" ON)
option(REQUEST_UDP_TRANSPORT "Request optional UDP endpoint adapter" OFF)
if(REQUEST_UDP_TRANSPORT)
    find_package(GameNetCore CONFIG REQUIRED COMPONENTS udp_transport)
elseif(REQUEST_UDP)
    find_package(GameNetCore CONFIG REQUIRED COMPONENTS udp)
    if(TARGET GameNet::udp_transport OR TARGET OpenSSL::SSL)
        message(FATAL_ERROR "UDP-only discovery loaded optional adapter/dependency")
    endif()
else()
    find_package(GameNetCore CONFIG REQUIRED)
    if(TARGET GameNet::udp OR TARGET GameNet::udp_transport OR TARGET OpenSSL::SSL)
        message(FATAL_ERROR "Core-only discovery loaded optional dependency")
    endif()
endif()
add_executable(udp_consumer main.cpp)
target_link_libraries(udp_consumer PRIVATE GameNet::core)
if(REQUEST_UDP OR REQUEST_UDP_TRANSPORT)
    target_link_libraries(udp_consumer PRIVATE GameNet::udp)
    target_compile_definitions(udp_consumer PRIVATE CONSUMER_UDP=1)
endif()
if(REQUEST_UDP_TRANSPORT)
    target_link_libraries(udp_consumer PRIVATE GameNet::udp_transport)
    target_compile_definitions(udp_consumer PRIVATE CONSUMER_UDP_TRANSPORT=1)
endif()
enable_testing()
add_test(NAME udp_installed_consumer COMMAND udp_consumer)
