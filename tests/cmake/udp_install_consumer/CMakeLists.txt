cmake_minimum_required(VERSION 3.20)
project(GameNetUdpConsumer LANGUAGES CXX)
option(REQUEST_UDP "Request optional UDP" ON)
if(REQUEST_UDP)
    find_package(GameNetCore CONFIG REQUIRED COMPONENTS udp)
else()
    find_package(GameNetCore CONFIG REQUIRED)
    if(TARGET GameNet::udp OR TARGET OpenSSL::SSL)
        message(FATAL_ERROR "Core-only discovery loaded optional dependency")
    endif()
endif()
add_executable(udp_consumer main.cpp)
target_link_libraries(udp_consumer PRIVATE GameNet::core)
if(REQUEST_UDP)
    target_link_libraries(udp_consumer PRIVATE GameNet::udp)
    target_compile_definitions(udp_consumer PRIVATE CONSUMER_UDP=1)
endif()
enable_testing()
add_test(NAME udp_installed_consumer COMMAND udp_consumer)
