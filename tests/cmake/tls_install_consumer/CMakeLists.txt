cmake_minimum_required(VERSION 3.20)
project(TlsInstallConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EXPECT_TLS "Request the optional TLS component" ON)
option(EXPECT_RPC_TLS "Request the optional RPC/TLS component" OFF)
option(EXPECT_RPC "Request RPC without TLS/OpenSSL" OFF)
if(EXPECT_RPC_TLS)
    find_package(GameNetCore REQUIRED COMPONENTS rpc_tls)
    add_executable(consumer rpc_tls.cpp)
    target_link_libraries(consumer PRIVATE GameNet::rpc_tls)
elseif(EXPECT_TLS)
    find_package(GameNetCore REQUIRED COMPONENTS tls)
    if(TARGET GameNet::rpc_tls)
        message(FATAL_ERROR "TLS-only discovery loaded the RPC/TLS adapter")
    endif()
    add_executable(consumer main.cpp)
    target_link_libraries(consumer PRIVATE GameNet::tls)
else()
    if(EXPECT_RPC)
        find_package(GameNetCore REQUIRED COMPONENTS rpc)
    else()
        find_package(GameNetCore REQUIRED)
    endif()
    if(TARGET GameNet::tls OR TARGET GameNet::rpc_tls OR TARGET OpenSSL::SSL OR TARGET OpenSSL::Crypto)
        message(FATAL_ERROR "Core-only discovery loaded optional TLS/OpenSSL")
    endif()
    if(EXPECT_RPC)
        add_executable(consumer rpc.cpp)
        target_link_libraries(consumer PRIVATE GameNet::rpc)
    else()
        add_executable(consumer core.cpp)
        target_link_libraries(consumer PRIVATE GameNet::core)
    endif()
endif()
enable_testing()
add_test(NAME installed.consumer COMMAND consumer)
