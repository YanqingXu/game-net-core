cmake_minimum_required(VERSION 3.20)
project(TlsInstallConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EXPECT_TLS "Request the optional TLS component" ON)
if(EXPECT_TLS)
    find_package(GameNetCore REQUIRED COMPONENTS tls)
    add_executable(consumer main.cpp)
    target_link_libraries(consumer PRIVATE GameNet::tls)
else()
    find_package(GameNetCore REQUIRED)
    if(TARGET GameNet::tls OR TARGET OpenSSL::SSL OR TARGET OpenSSL::Crypto)
        message(FATAL_ERROR "Core-only discovery loaded optional TLS/OpenSSL")
    endif()
    add_executable(consumer core.cpp)
    target_link_libraries(consumer PRIVATE GameNet::core)
endif()
enable_testing()
add_test(NAME installed.consumer COMMAND consumer)
